    sum += i

addite(1,100)更改

# 定義 addite 函數
def addite(start, end):
    sum = 0
    for i in range(start, end + 1):  # 使用 start 到 end (包括 end)
        sum += i
    return sum

# 使用 addite 函數來計算從 1 到 100 的總和
result = addite(1, 100)
print("累加結果是:", result)

add-only-even

# 定義 add_only_even 函式
def add_only_even(start, end):
    sum = 0
    for i in range(start, end + 1):  # 遍歷從 start 到 end (包括 end)
        if i % 2 == 0:  # 檢查是否為偶數
            sum += i
    return sum

# 測試 add_only_even 函式
result = add_only_even(1, 100)
print("1 到 100 的偶數累加結果是:", result)

add_avoid_8

# 定義 add_only_even 函式
def add_only_even(start, end):
    sum = 0
    for i in range(start, end + 1):  # 遍歷從 start 到 end (包括 end)
        if i % 2 == 0:  # 檢查是否為偶數
            sum += i
    return sum

# 定義 add_avoid_8 函式
def add_avoid_8(start, end):
    sum = 0
    for i in range(start, end + 1):  # 遍歷從 start 到 end (包括 end)
        if '8' not in str(i):  # 檢查數字中是否不包含"8"
            sum += i
    return sum

# 測試 add_only_even 函式
result = add_only_even(1, 100)
print("1 到 100 的偶數累加結果是:", result)
